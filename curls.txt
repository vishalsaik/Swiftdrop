# ---------------------------------------------------------------------------
# Full order lifecycle through the real endpoints:
#   create -> pay (fake gateway) -> accept (assigns a stub driver) -> ready -> pickup -> deliver
# ---------------------------------------------------------------------------

## 1. Create the order (starts as "created")
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"life-1","customerId":"cust-1","restaurantId":"rest-1","items":[{"sku":"SKU1","name":"Burger","quantity":1,"unitPriceCents":599}],"paymentType":"card"}'


## 2. Accept before paying — 409 (created cannot jump to preparing)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/accept


## 3. Pay (the fake gateway approves 599 cents) — 200, state "paid"
curl -i -X POST http://127.0.0.1:8080/orders/life-1/pay


## 4. Pay again — 200, a REPLAY of the first payment: the provider is not charged again
curl -i -X POST http://127.0.0.1:8080/orders/life-1/pay


## 5. Restaurant accepts — 200, state "preparing", assignedDriverId is set
curl -i -X POST http://127.0.0.1:8080/orders/life-1/accept


## 6. Pickup before the food is ready — 409 "Food is not ready yet"
curl -i -X POST http://127.0.0.1:8080/orders/life-1/pickup


## 7. Kitchen marks the food ready — 200, foodReady set, state stays "preparing"
curl -i -X POST http://127.0.0.1:8080/orders/life-1/ready


## 8. Ready again — 200, same foodReady timestamp (idempotent)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/ready


## 9. Pickup — 200, state "picked_up" (food ready AND driver assigned)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/pickup


## 10. Deliver — 200, state "delivered"
curl -i -X POST http://127.0.0.1:8080/orders/life-1/deliver


## 11. Deliver again — 409 (delivered is terminal)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/deliver


## Cancelling is allowed up to pickup, e.g. after paying
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"life-2","customerId":"cust-2","restaurantId":"rest-1","items":[{"sku":"SKU1","name":"Burger","quantity":1,"unitPriceCents":599}],"paymentType":"card"}'
curl -i -X POST http://127.0.0.1:8080/orders/life-2/pay
curl -i -X POST http://127.0.0.1:8080/orders/life-2/cancel


## Paying an order that was paid and then cancelled — 200, the first payment is replayed
## (the body shows state "cancelled"; refunds are not built yet)
curl -i -X POST http://127.0.0.1:8080/orders/life-2/pay

# ---------------------------------------------------------------------------
# Idempotent retries. /pay never charges twice for one order. These simulate the
# two states a retry can find a payment in, by writing the rows straight into
# Postgres (run them against the dev database; the ids must be new each time).
# A real payment gets its uuid at the claim; these use readable ids.
# ---------------------------------------------------------------------------

## Another request is working on the payment (a LIVE lease, one hour) — 409 "still being processed"
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"idem-live","customerId":"c","restaurantId":"r","paymentType":"card","items":[{"sku":"S","name":"Meal","quantity":1,"unitPriceCents":4000}]}'
docker exec postgres psql -U postgres -d swiftdrop -c "BEGIN; UPDATE orders SET state='payment_pending' WHERE id='idem-live'; INSERT INTO payments (id, order_id, status, amount_cents, lease_expires_at) VALUES ('live-pay', 'idem-live', 'processing', 4000, now() + interval '1 hour'); COMMIT;"
curl -i -X POST http://127.0.0.1:8080/orders/idem-live/pay


## The owner crashed (an EXPIRED lease): the retry takes the payment over and asks the
## provider again with the SAME payment id ("crashed-pay") — 200, state "paid"
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"idem-crash","customerId":"c","restaurantId":"r","paymentType":"card","items":[{"sku":"S","name":"Meal","quantity":1,"unitPriceCents":3000}]}'
docker exec postgres psql -U postgres -d swiftdrop -c "BEGIN; UPDATE orders SET state='payment_pending' WHERE id='idem-crash'; INSERT INTO payments (id, order_id, status, amount_cents, lease_expires_at) VALUES ('crashed-pay', 'idem-crash', 'processing', 3000, now() - interval '1 minute'); COMMIT;"
curl -i -X POST http://127.0.0.1:8080/orders/idem-crash/pay
docker exec postgres psql -U postgres -d swiftdrop -c "SELECT id, status, provider_payment_id FROM payments WHERE order_id='idem-crash';"


## Twenty simultaneous payments for one order: one payment row, one charge
## (bash; the late arrivals replay with 200, any that arrive mid-payment get 409)
curl -s -X POST http://127.0.0.1:8080/orders -H "Content-Type: application/json" \
  -d '{"id":"idem-race","customerId":"c","restaurantId":"r","paymentType":"card","items":[{"sku":"S","name":"Meal","quantity":1,"unitPriceCents":2000}]}' > /dev/null
for i in $(seq 1 20); do curl -s -o /dev/null -w "%{http_code}\n" -X POST http://127.0.0.1:8080/orders/idem-race/pay & done; wait
docker exec postgres psql -U postgres -d swiftdrop -c "SELECT count(*) AS payment_rows FROM payments WHERE order_id='idem-race';"


# ---------------------------------------------------------------------------
# CORS. Allowed origins: http://localhost:3000 and http://localhost:5173.
# curl is not a browser, so it never enforces CORS; these show the headers
# a browser would act on. Look for Access-Control-* and Vary in the output.
# ---------------------------------------------------------------------------

## Preflight from an allowed origin — 204 with Allow-Origin/Methods/Headers/Max-Age
curl -i -X OPTIONS http://127.0.0.1:8080/orders \
  -H "Origin: http://localhost:3000" \
  -H "Access-Control-Request-Method: POST" \
  -H "Access-Control-Request-Headers: Content-Type"


## Preflight from an origin that is not allowed — 405, no Access-Control-* headers
curl -i -X OPTIONS http://127.0.0.1:8080/orders \
  -H "Origin: http://evil.com" \
  -H "Access-Control-Request-Method: POST"


## Normal request from an allowed origin — response carries Access-Control-Allow-Origin
curl -i http://127.0.0.1:8080/orders/life-1 -H "Origin: http://localhost:5173"


## Error responses carry it too (415 here), so a browser can read the message
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Origin: http://localhost:5173" \
  -H "Content-Type: text/plain" \
  -d '{}'


## No Origin header (curl, server-to-server) — no Allow-Origin, but Vary: Origin is still set
curl -i http://127.0.0.1:8080/health
