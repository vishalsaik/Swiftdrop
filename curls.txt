# SwiftDrop API — manual test curls
# Server: go run ./cmd/api  (listens on 127.0.0.1:8080)
# DB: docker run --name postgres -e POSTGRES_PASSWORD=postgres -e POSTGRES_DB=swiftdrop -p 5434:5432 -d postgres:16-alpine

## Health check
curl -i http://127.0.0.1:8080/health


## Create order
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"order-1","customerId":"cust-1","restaurantId":"rest-1","items":[{"sku":"SKU1","name":"Burger","quantity":2,"unitPriceCents":599}],"paymentType":"card"}'


## Get order by id
curl -i http://127.0.0.1:8080/orders/order-1


## Get an order that was never created — 404
curl -i http://127.0.0.1:8080/orders/does-not-exist


## Create with an invalid body (missing/malformed JSON) — should 400
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d 'not valid json'


## Create a second order, to check list/lookup isn't accidentally sharing state
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"order-2","customerId":"cust-2","restaurantId":"rest-1","items":[{"sku":"SKU2","name":"Fries","quantity":1,"unitPriceCents":249}],"paymentType":"card"}'

curl -i http://127.0.0.1:8080/orders/order-2


## Cancel order — Created -> Cancelled is legal, should succeed (200)
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"order-3","customerId":"cust-3","restaurantId":"rest-1","items":[{"sku":"SKU3","name":"Pizza","quantity":1,"unitPriceCents":899}],"paymentType":"card"}'

curl -i -X POST http://127.0.0.1:8080/orders/order-3/cancel


## Cancel the same order again — Cancelled is terminal, should be rejected (409)
curl -i -X POST http://127.0.0.1:8080/orders/order-3/cancel


## Cancel an order that doesn't exist — 404
curl -i -X POST http://127.0.0.1:8080/orders/does-not-exist/cancel


# ---------------------------------------------------------------------------
# Request validation. POST /orders runs a middleware chain (body size,
# Content-Type) and then a validator chain (field rules); the first failure
# is returned as {"error": "..."}.
# ---------------------------------------------------------------------------

## A rule fails - 422 "paymentType must be one of: card, cash, wallet"
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"v-1","customerId":"c","restaurantId":"r","items":[{"sku":"S","name":"B","quantity":1,"unitPriceCents":100}]}'


## Missing id - 422 "id is required"
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"customerId":"c","restaurantId":"r","paymentType":"card","items":[{"sku":"S","name":"B","quantity":1,"unitPriceCents":100}]}'


## Empty cart - 422 "items must not be empty"
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"v-2","customerId":"c","restaurantId":"r","paymentType":"card","items":[]}'


## Bad quantity - 422 "items[0].quantity must be greater than 0"
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"v-3","customerId":"c","restaurantId":"r","paymentType":"card","items":[{"sku":"S","name":"B","quantity":0,"unitPriceCents":100}]}'


## Negative price - 422 "items[0].unitPriceCents must not be negative"
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"v-4","customerId":"c","restaurantId":"r","paymentType":"card","items":[{"sku":"S","name":"B","quantity":1,"unitPriceCents":-5}]}'


## Unknown field - 400 "invalid request body: json: unknown field ..."
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"v-5","bogus":1,"customerId":"c","restaurantId":"r","paymentType":"card","items":[{"sku":"S","name":"B","quantity":1,"unitPriceCents":100}]}'


## Wrong Content-Type - 415
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: text/plain" \
  -d '{"id":"v-6"}'


## Duplicate order id - 409 (run the same create twice)
curl -i -X POST http://127.0.0.1:8080/orders \

curl -i -X POST http://127.0.0.1:8080/orders \

  -d '{"id":"v-7","customerId":"c","restaurantId":"r","paymentType":"cash","items":[{"sku":"S","name":"B","quantity":1,"unitPriceCents":100}]}'

# ---------------------------------------------------------------------------
# Full order lifecycle through the real endpoints:
#   create -> pay (mock) -> accept (assigns a stub driver) -> ready -> pickup -> deliver
# ---------------------------------------------------------------------------

## 1. Create the order (starts as "created")
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"life-1","customerId":"cust-1","restaurantId":"rest-1","items":[{"sku":"SKU1","name":"Burger","quantity":1,"unitPriceCents":599}],"paymentType":"card"}'


## 2. Accept before paying — 409 (created cannot jump to preparing)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/accept


## 3. Pay (mock) — 200, state "paid"
curl -i -X POST http://127.0.0.1:8080/orders/life-1/pay


## 4. Pay again — 409 (already paid)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/pay


## 5. Restaurant accepts — 200, state "preparing", assignedDriverId is set
curl -i -X POST http://127.0.0.1:8080/orders/life-1/accept


## 6. Pickup before the food is ready — 409 "Food is not ready yet"
curl -i -X POST http://127.0.0.1:8080/orders/life-1/pickup


## 7. Kitchen marks the food ready — 200, foodReady set, state stays "preparing"
curl -i -X POST http://127.0.0.1:8080/orders/life-1/ready


## 8. Ready again — 200, same foodReady timestamp (idempotent)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/ready


## 9. Pickup — 200, state "picked_up" (food ready AND driver assigned)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/pickup


## 10. Deliver — 200, state "delivered"
curl -i -X POST http://127.0.0.1:8080/orders/life-1/deliver


## 11. Deliver again — 409 (delivered is terminal)
curl -i -X POST http://127.0.0.1:8080/orders/life-1/deliver


## Cancelling is allowed up to pickup, e.g. after paying
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Content-Type: application/json" \
  -d '{"id":"life-2","customerId":"cust-2","restaurantId":"rest-1","items":[{"sku":"SKU1","name":"Burger","quantity":1,"unitPriceCents":599}],"paymentType":"card"}'
curl -i -X POST http://127.0.0.1:8080/orders/life-2/pay
curl -i -X POST http://127.0.0.1:8080/orders/life-2/cancel


## Paying a cancelled order — 409
curl -i -X POST http://127.0.0.1:8080/orders/life-2/pay


## Any lifecycle endpoint on an order that doesn't exist — 404
curl -i -X POST http://127.0.0.1:8080/orders/does-not-exist/pay
curl -i -X POST http://127.0.0.1:8080/orders/does-not-exist/accept
curl -i -X POST http://127.0.0.1:8080/orders/does-not-exist/ready
curl -i -X POST http://127.0.0.1:8080/orders/does-not-exist/pickup
curl -i -X POST http://127.0.0.1:8080/orders/does-not-exist/deliver


# ---------------------------------------------------------------------------
# CORS. Allowed origins: http://localhost:3000 and http://localhost:5173.
# curl is not a browser, so it never enforces CORS; these show the headers
# a browser would act on. Look for Access-Control-* and Vary in the output.
# ---------------------------------------------------------------------------

## Preflight from an allowed origin — 204 with Allow-Origin/Methods/Headers/Max-Age
curl -i -X OPTIONS http://127.0.0.1:8080/orders \
  -H "Origin: http://localhost:3000" \
  -H "Access-Control-Request-Method: POST" \
  -H "Access-Control-Request-Headers: Content-Type"


## Preflight from an origin that is not allowed — 405, no Access-Control-* headers
curl -i -X OPTIONS http://127.0.0.1:8080/orders \
  -H "Origin: http://evil.com" \
  -H "Access-Control-Request-Method: POST"


## Normal request from an allowed origin — response carries Access-Control-Allow-Origin
curl -i http://127.0.0.1:8080/orders/life-1 -H "Origin: http://localhost:5173"


## Error responses carry it too (415 here), so a browser can read the message
curl -i -X POST http://127.0.0.1:8080/orders \
  -H "Origin: http://localhost:5173" \
  -H "Content-Type: text/plain" \
  -d '{}'


## No Origin header (curl, server-to-server) — no Allow-Origin, but Vary: Origin is still set
curl -i http://127.0.0.1:8080/health
